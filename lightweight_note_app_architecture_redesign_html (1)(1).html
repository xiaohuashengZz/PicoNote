不用手动按保存：Tiptap 的 onUpdate 回调 + Debounce = 500ms 自动保存。Tiptap 负责内容变化监听，Debounce 负责减少保存频率。

4. 数据库连接池

// 复用连接，不频繁打开关闭
// 读操作使用只读连接

pub struct DbPool {
    read_pool: Pool,
    write_pool: Pool,
}

impl DbPool {
    pub async fn query(&self, sql: &str) -> Result> {
        // 读操作走只读池
        self.read_pool.query(sql).await
    }

    pub async fn execute(&self, sql: &str) -> Result<()> {
        // 写操作走写池
        self.write_pool.execute(sql).await
    }
}

Repository 查询示例

// 相比手写 Repository，sea-orm 大幅简化
pub struct NoteRepository;

impl NoteRepository {
    pub async fn search(&self, db: &DatabaseConnection, keyword: &str) -> Result, DbErr> {
        Note::find()
            .filter(Column::Title.like(format!("%{}%", keyword)))
            .order_by(Column::UpdatedAt, Order::Desc)
            .limit(50)
            .all(db)
            .await
    }

    pub async fn list_by_workspace(&self, db: &DatabaseConnection, workspace_id: &str) -> Result, DbErr> {
        Note::find()
            .filter(Column::Status.eq(workspace_id.parse().unwrap_or(0)))
            .order_by(Column::IsPinned, Order::Desc)
            .order_by(Column::UpdatedAt, Order::Desc)
            .all(db)
            .await
    }
}

nanostores 原子化状态

// frontend/src/stores/noteStore.ts
import { atom, map } from 'nanostores';

// 笔记列表
export const $notes = atom([]);

// 当前选中的笔记 ID
export const $currentNoteId = atom(null);

// 当前笔记内容（派生状态）
export const $currentNote = computed($currentNoteId, (id) => {
  if (!id) return null;
  return $notes.get().find(n => n.id === id);
});

// 笔记操作
export const noteActions = {
  setNotes(notes: Note[]) {
    $notes.set(notes);
  },

  selectNote(id: string | null) {
    $currentNoteId.set(id);
  },

  createNote() {
    const note: Note = {
      id: crypto.randomUUID(),
      title: '新笔记',
      content: '',
      tags: [],
      isPinned: false,
      updatedAt: Date.now(),
    };
    $notes.set([note, ...$notes.get()]);
    $currentNoteId.set(note.id);
  },

  updateNote(id: string, updates: Partial) {
    $notes.set(
      $notes.get().map(n =>
        n.id === id ? { ...n, ...updates, updatedAt: Date.now() } : n
      )
    );
  },

  deleteNote(id: string) {
    $notes.set($notes.get().filter(n => n.id !== id));
    if ($currentNoteId.get() === id) {
      $currentNoteId.set(null);
    }
  },
};

// frontend/src/editor/Editor.ts
import { EditorState } from '@codemirror/state';
import { EditorView, basicSetup } from 'codemirror';
import { keymap } from '@codemirror/view';
import { defaultKeymap, history, historyKeymap } from '@codemirror/commands';

export const createEditor = (parent: HTMLElement, initial: string) => {
  const view = new EditorView({
    state: EditorState.create({
      doc: initial,
      extensions: [
        basicSetup,
        history(),
        keymap.of([...defaultKeymap, ...historyKeymap]),
        EditorView.lineWrapping,
        EditorView.updateListener.of((update) => {
          if (update.docChanged) {
            debounce(() => save(update.state.doc.toString()), 300);
          }
        }),
      ],
    }),
    parent,
  });
  return view;
};

// 销毁时释放内存
export const destroyEditor = (view: EditorView) => {
  view.destroy();
  view = null;
};

自动保存流程

1. 用户输入 → CodeMirror 更新 → updateListener 触发
2. Debounce 计时器重置（300ms）
3. 300ms 无新输入，触发保存
4. 调用 Tauri Command: invoke('save_note', { id, content })
5. Rust 侧：sea-orm 更新数据库

// debounce 实现
const debounce = (fn: Function, ms: number) => {
  let timer: number;
  return (...args: any[]) => {
    clearTimeout(timer);
    timer = setTimeout(() => fn(...args), ms);
  };
};

Rust 后端事件（nanoevents）

// src-tauri/src/events/emitter.rs
use nanoemma::EventEmitter;
use std::sync::Arc;

#[derive(Clone, Event)]
pub enum AppEvent {
    #[event("note:created")]
    NoteCreated { note_id: String },

    #[event("note:updated")]
    NoteUpdated { note_id: String, fields: Vec },

    #[event("note:deleted")]
    NoteDeleted { note_id: String },

    #[event("block:created")]
    BlockCreated { note_id: String, block_id: String },

    #[event("block:updated")]
    BlockUpdated { block_id: String },

    #[event("block:deleted")]
    BlockDeleted { note_id: String, block_id: String },

    #[event("tag:created")]
    TagCreated { tag_id: String, name: String },

    #[event("tag:deleted")]
    TagDeleted { tag_id: String },

    #[event("workspace:changed")]
    WorkspaceChanged { workspace_id: String },

    #[event("theme:changed")]
    ThemeChanged { theme: String },
}

// 使用
pub struct NoteService {
    emitter: Arc>,
}

impl NoteService {
    pub fn create_note(&self, title: &str) -> Result {
        let note = self.repo.create(title)?;
        self.emitter.emit(AppEvent::NoteCreated { note_id: note.id.clone() });
        Ok(note)
    }
}

// frontend/src/lib/events.ts
import { createNanoEvent } from 'nanoevents';

interface AppEvents {
  'note:created': (note: Note) => void;
  'note:updated': (note: Note) => void;
  'note:deleted': (id: string) => void;
  'theme:changed': (theme: 'light' | 'dark') => void;
}

export const emitter = createNanoEvent();

// 订阅
const unsub = emitter.on('note:updated', (note) => {
  console.log('Note updated:', note);
});

// 触发（通过 Tauri 事件 bridge）
import { listen } from '@tauri-apps/api/event';
listen('note:updated', (event) => {
  emitter.emit('note:updated', event.payload as Note);
});

// 取消订阅
unsub();

// React 组件中使用
useEffect(() => {
  const unsub = emitter.on('theme:changed', (theme) => {
    document.body.dataset.theme = theme;
  });
  return unsub;
}, []);

Rust Command 实现示例

// src-tauri/src/application/commands/note_commands.rs
use tauri::command;
use sea_orm::DatabaseConnection;

#[command]
pub async fn create_note(
    db: State<'_, DatabaseConnection>,
    emitter: EventEmitter,
    title: String,
) -> Result {
    let note = NoteService::new(db).create(&title)
        .map_err(|e| e.to_string())?;

    emitter.emit(AppEvent::NoteCreated { note_id: note.id.clone() });
    Ok(note)
}

#[command]
pub async fn search_notes(
    db: State<'_, DatabaseConnection>,
    keyword: String,
    limit: Option,
) -> Result, String> {
    NoteService::new(db)
        .search(&keyword, limit.unwrap_or(20))
        .await
        .map_err(|e| e.to_string())
}

// 注册命令
// src-tauri/src/main.rs
fn main() {
    tauri::Builder::default()
        .manage(db.clone())
        .invoke_handler(tauri::generate_handler![
            note_commands::create_note,
            note_commands::update_note,
            note_commands::delete_note,
            note_commands::get_note,
            note_commands::search_notes,
            note_commands::list_notes,
            tag_commands::create_tag,
            tag_commands::delete_tag,
        ])
        .run(tauri::generate_context!())
        .expect("error while running tauri application");
}

前端调用示例

// frontend/src/lib/tauri.ts
import { invoke } from '@tauri-apps/api/core';

export const noteApi = {
  create: (title: string) => invoke('create_note', { title }),
  update: (id: string, data: Partial) => invoke('update_note', { id, ...data }),
  delete: (id: string) => invoke('delete_note', { id }),
  get: (id: string) => invoke('get_note', { id }),
  search: (keyword: string, limit?: number) => invoke('search_notes', { keyword, limit }),
  list: (workspaceId?: string, offset = 0, limit = 50) =>
    invoke('list_notes', { workspace_id: workspaceId, offset, limit }),
};

// React Hook 封装
export const useNoteApi = () => {
  const setNotes = useNoteStore(s => s.setNotes);

  const createNote = async (title: string) => {
    const note = await noteApi.create(title);
    setNotes(prev => [note, ...prev]);
    return note;
  };

  return { createNote, ...noteApi };
};
